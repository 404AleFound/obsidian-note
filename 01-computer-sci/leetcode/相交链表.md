# 题目

给你两个单词 `word1` 和 `word2`， _请返回将 `word1` 转换成 `word2` 所使用的最少操作数_  。

你可以对一个单词进行如下三种操作：

- 插入一个字符
- 删除一个字符
- 替换一个字符

**示例 1：**

**输入：** `word1 = "horse", word2 = "ros"`
**输出：** `3`

# 题解

```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
    ListNode *getIntersectionNode(ListNode *headA, ListNode *headB) {
        int lenA = 0, lenB = 0;
        ListNode *pA = headA, *pB = headB;
        while(pA) {
            pA = pA->next;
            lenA++;
        }
        
        while(pB) {
            pB = pB->next;
            lenB++;
        }
        
        int lenCurr = abs(lenA - lenB);
        if (lenA > lenB) {
            while(lenCurr--) {
                headA = headA->next;
            }
        } else {
            while(lenCurr--) {
                headB = headB->next;
            }
        }
        
        while(headA && headB) {
            if (headA == headB) return headA;
            headA = headA->next;
            headB = headB->next;
        }
        return nullptr;
        
    }
};
```
